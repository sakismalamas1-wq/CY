function savePrintConfig() {
    const val = document.getElementById('print-copies').value;
    const saved = JSON.parse(localStorage.getItem('gbr_ui_settings') || "{}");
    saved['--print-copies'] = val;
    localStorage.setItem('gbr_ui_settings', JSON.stringify(saved));
    alert("Η ρύθμιση αποθηκεύτηκε: " + val + " αντίγραφα.");
}

/* ==========================================================================
   ΠΡΑΣΙΝΟ ΣΧΟΛΙΟ: ΕΚΤΥΠΩΣΗ ΜΕ ΑΥΤΟΜΑΤΟ ΚΟΨΙΜΟ (CUT) ΑΝΑ ΑΝΤΙΓΡΑΦΟ

async function printToKitchen() {
    const saved = JSON.parse(localStorage.getItem('gbr_ui_settings') || "{}");
    const printFS = saved['--print-font-size'] || '22px'; 
    const copies = parseInt(saved['--print-copies']) || 1; 

    let newItems = currentOrder.filter(it => !it.printed);
    let oldItems = currentOrder.filter(it => it.printed); 
    if (newItems.length === 0) return alert("Όλα εκτυπωμένα!");

    let sorted = sortOrderItems([...newItems]);
    
    let printContent = `<html><head><style>
        @media print {
            .ticket { page-break-after: always; } /* ΕΔΩ ΔΙΝΕΙ ΕΝΤΟΛΗ ΓΙΑ ΚΟΨΙΜΟ */
            body { margin: 0; }
        }
    </style></head><body style="font-family:Arial; width:80mm; padding:0; margin:0;">`;
    
    // ΦΤΙΑΧΝΟΥΜΕ ΤΟ ΣΩΜΑ ΤΗΣ ΑΠΟΔΕΙΞΗΣ
    let receiptBody = `<div class="ticket" style="padding:10px;">`;
    receiptBody += `<br><br>`;
    
    if(oldItems.length > 0) {
        receiptBody += `<center><h1 style="background:#000; color:#fff; padding:5px; border:2px solid #000;">* ΣΥΜΠΛΗΡΩΜΑ *</h1></center>`;
    }
    receiptBody += `<center><h2>${selectedTable}</h2><hr></center>`;
    
    // ... (εδώ μένει ο κώδικας για τα groups και τα counts όπως τον είχαμε) ...
    // [Ο κώδικας που μετράει τα counts και uniqueNames]
    let lastCatGroup = "";
    let counts = {};
    sorted.forEach(it => { counts[it.n] = (counts[it.n] || 0) + 1; });
    let uniqueNames = [...new Set(sorted.map(it => it.n))];

    uniqueNames.forEach(name => {
        // ... (το loop για τα ονόματα και τα groups) ...
        receiptBody += `<div style="font-size:${printFS}; font-weight:bold; padding:5px 0;">${counts[name]} x ${name}</div>`;
    });

    receiptBody += `<br><br><br><br><br><br></div>`; // Έξτρα κενό για να προλάβει να βγει το χαρτί πριν κόψει

    // ΠΡΟΣΘΗΚΗ ΑΝΤΙΓΡΑΦΩΝ
    for (let i = 0; i < copies; i++) {
        printContent += receiptBody;
    }

    printContent += `</body></html>`;

    const win = window.open('', '', 'width=600,height=800');
    win.document.write(printContent);
    
    // ΠΡΑΣΙΝΟ ΣΧΟΛΙΟ: ΑΥΤΟΜΑΤΟ PRINT & CLOSE
    win.document.write(`<script>
        window.onload = function() { 
            window.print(); 
            setTimeout(function(){ window.close(); }, 500); 
        };
    <\/script>`);
    win.document.close();

    currentOrder.forEach(it => it.printed = true);
    await save();
    renderOrder();
}

2. Η Ρύθμιση στον Driver του Εκτυπωτή (Πολύ Σημαντικό!)
Για να δουλέψει το παραπάνω και να "πέφτουν" τα χαρτάκια κομμένα, πρέπει να κάνεις μια ρύθμιση στα Windows:

Πήγαινε Πίνακας Ελέγχου > Συσκευές και Εκτυπωτές.

Δεξί κλικ στον θερμικό σου εκτυπωτή > Ιδιότητες Εκτυπωτή (Printer Properties).

Πήγαινε στην καρτέλα Device Settings (ή Advanced).

Ψάξε για μια επιλογή που λέει Paper Cut ή Cutter.

Επίλεξε το "Report [Cut]" ή "Document [Cut]".

3. Μόνιμη επιλογή στο Admin
Επειδή θέλεις να είναι μόνιμο, στο Admin θα το βάλεις μία φορά στο "2" ή στο "3" και δεν θα το ξαναπειράξεις. Ο κώδικας parseInt(saved['--print-copies']) || 1 θα διαβάζει αυτή τη ρύθμιση κάθε φορά που πατάς το κουμπί "ΚΟΥΖΙΝΑ".

Αν το κάνεις αυτό:

Πατάς "ΚΟΥΖΙΝΑ".

Ο εκτυπωτής βγάζει το 1ο χαρτάκι -> ΚΛΑΚ (Κόβει).

Βγάζει το 2ο χαρτάκι -> ΚΛΑΚ (Κόβει).

Καθαρά χέρια, καθαρές αποδείξεις!
